Google Apps Script (GAS) 後端部署教學與代碼

零伺服器架構，與 Google 試算表直接連動

1 Google 試算表欄位設定

建立一份新的 Google Sheet，第 1 列請確保標題名稱如下（不可錯字）：

Token | 姓名 | 電話 | 桌次備註 | 報到狀態 | 報到時間

2 複製貼上 Code.gs 程式碼

在試算表點選「擴充功能」>「Apps Script」，清空後貼上以下程式：

/**
 * 課程快速報到系統後端 API (Google Apps Script)
 * Sheet 結構: [A: Token, B: 姓名, C: 電話, D: 桌次備註, E: 報到狀態, F: 報到時間]
 */

function doGet(e) {
  return handleRequest(e);
}

function doPost(e) {
  return handleRequest(e);
}

function handleRequest(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000); // 避免多個志工同時掃描發生寫入衝突

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getActiveSheet();
    var data = sheet.getDataRange().getValues();
    
    var action = e.parameter.action || (e.postData ? JSON.parse(e.postData.contents).action : "list");
    
    // 1. 取得所有學員名單
    if (action === "list") {
      var attendees = [];
      for (var i = 1; i < data.length; i++) {
        attendees.push({
          token: String(data[i][0]),
          name: String(data[i][1]),
          phone: String(data[i][2]),
          note: String(data[i][3]),
          status: String(data[i][4]) || "pending",
          checkinTime: data[i][5] ? Utilities.formatDate(new Date(data[i][5]), "Asia/Taipei", "HH:mm:ss") : ""
        });
      }
      return jsonResponse({ success: true, attendees: attendees });
    }

    // 2. 進行核銷簽到
    if (action === "checkin") {
      var token = e.parameter.token || (e.postData ? JSON.parse(e.postData.contents).token : "");
      if (!token) return jsonResponse({ success: false, message: "缺少票券 Token" });

      for (var r = 1; r < data.length; r++) {
        if (String(data[r][0]).trim() === String(token).trim()) {
          var currentStatus = String(data[r][4]);
          if (currentStatus === "attended") {
            return jsonResponse({
              success: false,
              alreadyAttended: true,
              message: "重複報到！該學員已於 " + (data[r][5] ? Utilities.formatDate(new Date(data[r][5]), "Asia/Taipei", "HH:mm:ss") : "稍早") + " 完成報到",
              attendee: { name: data[r][1], note: data[r][3] }
            });
          }

          // 標記出席並寫入時間
          var now = new Date();
          sheet.getRange(r + 1, 5).setValue("attended");
          sheet.getRange(r + 1, 6).setValue(now);

          return jsonResponse({
            success: true,
            message: "報到成功！",
            attendee: {
              token: data[r][0],
              name: data[r][1],
              phone: data[r][2],
              note: data[r][3],
              status: "attended",
              checkinTime: Utilities.formatDate(now, "Asia/Taipei", "HH:mm:ss")
            }
          });
        }
      }
      return jsonResponse({ success: false, notFound: true, message: "查無此票券代碼，無效票券！" });
    }

    return jsonResponse({ success: false, message: "未知操作" });

  } catch (err) {
    return jsonResponse({ success: false, error: err.toString() });
  } finally {
    lock.releaseLock();
  }
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

3 發布為網頁應用程式 (Web App)

  1. 點擊右上角「部署」按鈕 >「新增部署作業」。
  2. 類型選擇「網頁應用程式 (Web App)」。
  3. 誰可以存取 (Who has access)：務必選 「所有人 (Anyone)」（這樣前端瀏覽器才有權限核銷）。
  4. 點擊「部署」，複製產生的「網頁應用程式網址 (Web App URL)」。
  5. 回到本網頁點擊右上角「GAS 試算表設定」，將 URL 貼入即可啟用真機串接！
